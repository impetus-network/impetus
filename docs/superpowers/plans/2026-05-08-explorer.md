# Artemis Explorer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an EVM block explorer for the Artemis chain at `packages/explorer` using Next.js 16 + tRPC v11 + Prisma (read-only) + coss UI primitives.

**Architecture:** Prisma introspects Subscan's existing Postgres. tRPC routers convert Prisma types to wire-friendly JSON. Next.js 16 Server Components call tRPC server-side caller; client islands handle polling and interactivity. Shared `@artemis/coss-ui` package provides Base-UI primitives to both `packages/ui` and `packages/explorer`.

**Tech Stack:** Next.js 16, React 19, tRPC v11, Prisma (introspect-only), Tailwind v4, @base-ui/react, date-fns, viem, Vitest, Playwright

**Design spec:** `docs/superpowers/specs/2026-05-07-explorer-design.md`

- [ ] **Step 4: Implement formatBalance**

Create `packages/shared/src/format/balance.ts`:

```ts
import { formatUnits } from "viem";

export function formatBalance(
  wei: bigint | string,
  decimals: number,
  displayDecimals = 4,
): string {
  const raw = formatUnits(typeof wei === "string" ? BigInt(wei) : wei, decimals);
  const [int, frac = ""] = raw.split(".");
  const intGrouped = new Intl.NumberFormat("en-US").format(BigInt(int));
  const fracTrimmed = frac.slice(0, displayDecimals).replace(/0+$/, "");
  return fracTrimmed ? `${intGrouped}.${fracTrimmed}` : intGrouped;
}
```

```ts
export function formatTimestamp(unixSec: number): string {
  const fmt = new Intl.DateTimeFormat("en-US", {
    dateStyle: "medium",
    timeStyle: "medium",
    timeZone: "UTC",
  });
  return `${fmt.format(new Date(unixSec * 1000))} UTC`;
}
```

```ts
import { getAddress } from "viem";

export function hex(b: Buffer): `0x${string}` {
  return `0x${b.toString("hex")}` as `0x${string}`;
}

export function addr(b: Buffer): `0x${string}` {
  return getAddress(hex(b));
}

export function unix(d: Date): number {
  return Math.floor(d.getTime() / 1000);
}

export function safeNum(n: bigint): number {
  if (n > BigInt(Number.MAX_SAFE_INTEGER)) {
    throw new Error(`safeNum overflow: ${n}`);
  }
  return Number(n);
}

export function bigStr(n: bigint | { toFixed: (dp: number) => string }): string {
  return typeof n === "bigint" ? n.toString() : n.toFixed(0);
}

export function nullable<T, U>(
  x: T | null,
  fn: (t: T) => U,
): U | null {
  return x === null ? null : fn(x);
}
```

```ts
export type SearchCategory = "block_num" | "tx_hash" | "address" | "unknown";

export function classify(q: string): SearchCategory {
  const trimmed = q.trim();
  if (/^\d+$/.test(trimmed)) return "block_num";
  if (/^0x[0-9a-fA-F]{64}$/.test(trimmed)) return "tx_hash";
  if (/^0x[0-9a-fA-F]{40}$/.test(trimmed)) return "address";
  return "unknown";
}
```

```ts
import { z } from "zod";
import { publicProcedure, router } from "../trpc";
import { addr, hex, safeNum, unix } from "../lib/encode";
import { decodeCursor, encodeCursor } from "../lib/cursor";

interface BlockCursor {
  id: number;
}

export const blockRouter = router({
  latest: publicProcedure
    .input(z.object({ limit: z.number().int().min(1).max(20).default(10) }))
    .query(async ({ ctx, input }) => {
      const rows = await ctx.prisma.evm_block.findMany({
        orderBy: { block_num: "desc" },
        take: input.limit,
      });
      return rows.map((r) => ({
        blockNum: safeNum(r.block_num),
        hash: hex(r.hash),
        miner: addr(r.miner),
        timestamp: unix(r.block_timestamp),
        txCount: r.transaction_count,
      }));
    }),

  list: publicProcedure
    .input(
      z.object({
        cursor: z.string().optional(),
        limit: z.number().int().min(1).max(50).default(25),
      }),
    )
    .query(async ({ ctx, input }) => {
      const { limit } = input;
      const cursor = input.cursor
        ? decodeCursor<BlockCursor>(input.cursor)
        : undefined;

      const rows = await ctx.prisma.evm_block.findMany({
        orderBy: { id: "desc" },
        take: limit + 1,
        ...(cursor ? { cursor: { id: cursor.id }, skip: 1 } : {}),
      });

      const hasMore = rows.length > limit;
      const items = (hasMore ? rows.slice(0, limit) : rows).map((r) => ({
        blockNum: safeNum(r.block_num),
        hash: hex(r.hash),
        miner: addr(r.miner),
        timestamp: unix(r.block_timestamp),
        txCount: r.transaction_count,
      }));

      return {
        items,
        nextCursor: hasMore
          ? encodeCursor({ id: rows[limit - 1].id })
          : null,
        prevCursor: cursor ? encodeCursor({ id: cursor.id }) : null,
      };
    }),

  byNumber: publicProcedure
    .input(z.object({ num: z.coerce.number().int().min(0) }))
    .query(async ({ ctx, input }) => {
      const row = await ctx.prisma.evm_block.findFirst({
        where: { block_num: BigInt(input.num) },
      });
      if (!row) return null;

      return {
        blockNum: safeNum(row.block_num),
        hash: hex(row.hash),
        parentHash: hex(row.parent_hash),
        miner: addr(row.miner),
        timestamp: unix(row.block_timestamp),
        txCount: row.transaction_count,
        size: row.size,
        gasUsed: safeNum(row.gas_used),
        gasLimit: safeNum(row.gas_limit),
      };
    }),

  txList: publicProcedure
    .input(
      z.object({
        blockNum: z.number().int(),
        cursor: z.string().optional(),
        limit: z.number().int().min(1).max(50).default(25),
      }),
    )
    .query(async ({ ctx, input }) => {
      const cursor = input.cursor
        ? decodeCursor<{ id: number }>(input.cursor)
        : undefined;

      const rows = await ctx.prisma.evm_transaction.findMany({
        where: { block_num: BigInt(input.blockNum) },
        orderBy: { id: "asc" },
        take: input.limit + 1,
        ...(cursor ? { cursor: { id: cursor.id }, skip: 1 } : {}),
      });

      const hasMore = rows.length > input.limit;
      const items = (hasMore ? rows.slice(0, input.limit) : rows).map(
        (r) => ({
          hash: hex(r.hash),
          blockNum: safeNum(r.block_num),
          from: addr(r.from_address),
          to: r.to_address ? addr(r.to_address) : null,
          value: r.value.toFixed(0),
          timestamp: unix(r.block_timestamp),
        }),
      );

      return {
        items,
        nextCursor: hasMore
          ? encodeCursor({ id: rows[input.limit - 1].id })
          : null,
        prevCursor: cursor ? encodeCursor({ id: cursor.id }) : null,
      };
    }),
});
```

- [ ] **Step 6: Implement search router**

Create `packages/explorer/src/server/routers/search.ts`:

```ts
import { z } from "zod";
import { publicProcedure, router } from "../trpc";
import { addr, hex } from "../lib/encode";
import { classify } from "@/lib/regex";

type SearchResult =
  | { kind: "block"; num: number }
  | { kind: "tx"; hash: `0x${string}` }
  | { kind: "address"; addr: `0x${string}` }
  | { kind: "contract"; addr: `0x${string}` }
  | { kind: "not_found" };

export const searchRouter = router({
  resolve: publicProcedure
    .input(z.object({ query: z.string().min(1).max(200) }))
    .query(async ({ ctx, input }): Promise<SearchResult> => {
      const category = classify(input.query);

      switch (category) {
        case "block_num": {
          const num = parseInt(input.query.trim(), 10);
          const exists = await ctx.prisma.evm_block.findFirst({
            where: { block_num: BigInt(num) },
            select: { id: true },
          });
          return exists ? { kind: "block", num } : { kind: "not_found" };
        }

        case "tx_hash": {
          const hashBuf = Buffer.from(input.query.trim().slice(2), "hex");
          const row = await ctx.prisma.evm_transaction.findFirst({
            where: { hash: hashBuf },
            select: { hash: true },
          });
          return row
            ? { kind: "tx", hash: hex(row.hash) }
            : { kind: "not_found" };
        }

        case "address": {
          const addrBuf = Buffer.from(input.query.trim().slice(2), "hex");
          const account = await ctx.prisma.evm_account.findFirst({
            where: { address: addrBuf },
            select: { address: true, code: true },
          });

          if (!account) return { kind: "not_found" };

          const hasCode =
            account.code !== null &&
            account.code.length > 0 &&
            account.code.toString("hex") !== "" &&
            account.code.toString("hex") !== "00";

          return hasCode
            ? { kind: "contract", addr: addr(account.address) }
            : { kind: "address", addr: addr(account.address) };
        }

        default:
          return { kind: "not_found" };
      }
    }),
});
```
